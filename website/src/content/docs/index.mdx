import { Card, CardGrid, LinkCard, Code } from '@astrojs/starlight/components';

export const example = `import ChatBot, { type Step } from 'react-simple-chatbot';

const steps: Step[] = [
  { id: 'ask', message: 'What is your name?', trigger: 'name' },
  { id: 'name', user: true, trigger: 'greet' },
  { id: 'greet', message: 'Hi {previousValue}, nice to meet you!', end: true }
];

export default function App() {
  return <ChatBot steps={steps} />;
}`;

## Why React Simple Chatbot

<CardGrid>
  <Card title="Conversations as data" icon="list-format">
    A conversation is a plain array of steps: bot messages, user inputs, options and custom components, connected by triggers.
  </Card>
  <Card title="TypeScript first" icon="seti:typescript">
    Written in TypeScript with types for every step and prop, so your editor completes and checks your conversations.
  </Card>
  <Card title="Your components inside" icon="puzzle">
    Render any React component as a step, read the previous answers, and decide when the conversation continues.
  </Card>
  <Card title="Themeable" icon="seti:css">
    Change colors and fonts with a styled-components theme, or style any element with props and stable class names.
  </Card>
  <Card title="Voice ready" icon="comment-alt">
    Speech recognition for user answers and speech synthesis for bot messages, with one prop each.
  </Card>
  <Card title="Production ready" icon="approve-check">
    React 18 and 19, StrictMode, server-side rendering, conversation cache and accessible controls.
  </Card>
</CardGrid>
